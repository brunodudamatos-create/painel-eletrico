let dadosGlobais = null; let chartMensal = null; let chartDiario = null; let chartSaldo = null; // ================================================================ // FORMATADORES // ================================================================ function numero(valor) { if ( valor === null || valor === undefined || !Number.isFinite(Number(valor)) ) { return null; } return Number(valor); } function formatarKwh(valor) { const n = numero(valor); if (n === null) { return '—'; } return `${n.toFixed(2)} kWh`; } function formatarRs(valor) { const n = numero(valor); if (n === null) { return '—'; } return `R$ ${n.toFixed(2)}`; } // ================================================================ // ALTERA TÍTULOS DOS CARDS EXISTENTES // ================================================================ function corrigirTitulosCards() { const textos = document.body.querySelectorAll('*'); textos.forEach(elemento => { if ( elemento.children.length === 0 && elemento.textContent ) { const texto = elemento.textContent.trim(); if ( texto === 'GERAÇÃO SOLAR DO MÊS' || texto === 'GERAÇÃO SOLAR' ) { elemento.textContent = 'ENERGIA EXPORTADA PARA REDE'; } if ( texto === 'CONSUMO TOTAL DO MÊS' || texto === 'CONSUMO TOTAL' ) { elemento.textContent = 'CONSUMO DA REDE'; } if ( texto === 'SALDO DO MÊS (INJETADO - CONSUMIDO)' || texto === 'SALDO DO MÊS' ) { elemento.textContent = 'GERAÇÃO FV'; } if ( texto === 'ECONOMIA NO MÊS (R$)' || texto === 'ECONOMIA NO MÊS' ) { elemento.textContent = 'ECONOMIA SOLAR'; } } }); } // ================================================================ // CARREGAMENTO // ================================================================ async function carregarGestaoEnergetica() { try { const res = await fetch( '/api/gestao?v=' + new Date().getTime() ); dadosGlobais = await res.json(); if (dadosGlobais.erro) { alert( 'Erro na API: ' + dadosGlobais.erro ); return; } if ( !dadosGlobais.mensais || dadosGlobais.mensais.length === 0 ) { console.warn( 'Nenhum dado encontrado.' ); return; } const selectMes = document.getElementById( 'seletorMes' ); if (!selectMes) { return; } selectMes.innerHTML = ''; dadosGlobais.mensais.forEach(m => { const option = document.createElement('option'); option.value = m.mes; const partes = m.mes.split('-'); option.text = `${partes[1]}/${partes[0]}`; selectMes.appendChild(option); }); selectMes.selectedIndex = selectMes.options.length - 1; selectMes.onchange = function () { renderizarMesSelecionado( this.value ); }; renderizarMesSelecionado( selectMes.value ); renderizarGraficoMensal( dadosGlobais.mensais ); corrigirTitulosCards(); } catch (erro) { console.error( 'Falha ao carregar gestão:', erro ); } } // ================================================================ // MÊS // ================================================================ function renderizarMesSelecionado(mesStr) { const mesObj = dadosGlobais.mensais.find( m => m.mes === mesStr ); if (!mesObj) { return; } const partesMes = mesStr.split('-') .reverse() .join('/'); const titulo = document.getElementById( 'titulo-mes' ); if (titulo) { titulo.innerText = `RESUMO DO MÊS (${partesMes})`; } // ============================================================ // CARDS // ============================================================ // Exportação Tuya const mesGeracao = document.getElementById( 'mes-geracao' ); if (mesGeracao) { mesGeracao.innerText = formatarKwh( mesObj.energia_exportada_kwh ); } // Consumo da rede const mesConsumo = document.getElementById( 'mes-consumo' ); if (mesConsumo) { mesConsumo.innerText = formatarKwh( mesObj.consumo_rede_kwh ); } // Geração FV const mesSaldo = document.getElementById( 'mes-saldo' ); if (mesSaldo) { mesSaldo.innerText = 'Aguardando Elekeeper'; mesSaldo.style.color = '#8b949e'; } // Economia const mesEconomia = document.getElementById( 'mes-economia' ); if (mesEconomia) { mesEconomia.innerText = 'Aguardando geração FV'; mesEconomia.style.color = '#8b949e'; } // ============================================================ // INDICADORES // ============================================================ const diarios = dadosGlobais.diarios .filter( d => d.data.startsWith( mesStr ) ); if (diarios.length > 0) { const hoje = diarios[diarios.length - 1]; const ultimos7 = diarios.slice(-7); // -------------------------------------------------------- // Exportação dos últimos 7 dias // -------------------------------------------------------- const exportacao7 = ultimos7.reduce( (total, item) => total + Number( item.energia_exportada_kwh || 0 ), 0 ); const hojeSaldo = document.getElementById( 'hoje-saldo' ); if (hojeSaldo) { hojeSaldo.innerText = `Exportação: ${ Number( hoje.energia_exportada_kwh || 0 ).toFixed(2) } kWh`; hojeSaldo.style.color = '#3fb950'; } const semanaSaldo = document.getElementById( 'semana-saldo' ); if (semanaSaldo) { semanaSaldo.innerText = `Exportação: ${ exportacao7.toFixed(2) } kWh`; semanaSaldo.style.color = '#3fb950'; } } // ============================================================ // GRÁFICOS // ============================================================ renderizarGraficosDiarios( diarios ); } // ================================================================ // GRÁFICO MENSAL // ================================================================ function renderizarGraficoMensal(mensais) { const canvas = document.getElementById( 'graficoMensal' ); if (!canvas) { return; } const ctx = canvas.getContext('2d'); const labels = mensais.map( m => m.mes .split('-') .reverse() .join('/') ); if (chartMensal) { chartMensal.destroy(); } chartMensal = new Chart(ctx, { type: 'bar', data: { labels, datasets: [ { label: 'Exportação para rede (kWh)', data: mensais.map( m => Number( m.energia_exportada_kwh || 0 ) ), backgroundColor: '#f59e0b' }, { label: 'Consumo da rede (kWh)', data: mensais.map( m => Number( m.consumo_rede_kwh || 0 ) ), backgroundColor: '#f85149' } ] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true, grid: { color: '#30363d' }, ticks: { color: '#8b949e' } }, x: { grid: { display: false }, ticks: { color: '#8b949e' } } }, plugins: { legend: { labels: { color: '#e6edf3' } } } } }); } // ================================================================ // GRÁFICOS DIÁRIOS // ================================================================ function renderizarGraficosDiarios(dados) { const labels = dados.map( d => { const partes = d.data.split('-'); return `${partes[2]}/${partes[1]}`; } ); // ============================================================ // GRÁFICO 1 // CONSUMO DA REDE X EXPORTAÇÃO // ============================================================ const canvasDiario = document.getElementById( 'graficoDiario' ); if (canvasDiario) { if (chartDiario) { chartDiario.destroy(); } chartDiario = new Chart( canvasDiario.getContext('2d'), { type: 'bar', data: { labels, datasets: [ { label: 'Consumo da rede (kWh)', data: dados.map( d => Number( d.consumo_rede_kwh || 0 ) ), backgroundColor: '#f85149' }, { label: 'Exportação para rede (kWh)', data: dados.map( d => Number( d.energia_exportada_kwh || 0 ) ), backgroundColor: '#f59e0b' } ] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true, grid: { color: '#30363d' }, ticks: { color: '#8b949e' } }, x: { grid: { display: false }, ticks: { color: '#8b949e' } } }, plugins: { legend: { labels: { color: '#e6edf3' } } } } } ); } // ============================================================ // GRÁFICO 2 // NÃO MOSTRA MAIS "BALANÇO" FALSO // Mostra exportação diária. // Quando Elekeeper entrar, este gráfico será substituído // pelo balanço real de energia. // ============================================================ const canvasSaldo = document.getElementById( 'graficoSaldo' ); if (canvasSaldo) { if (chartSaldo) { chartSaldo.destroy(); } chartSaldo = new Chart( canvasSaldo.getContext('2d'), { type: 'bar', data: { labels, datasets: [ { label: 'Exportação para rede (kWh)', data: dados.map( d => Number( d.energia_exportada_kwh || 0 ) ), backgroundColor: '#f59e0b' } ] }, options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true, grid: { color: '#30363d' }, ticks: { color: '#8b949e' } }, x: { grid: { display: false }, ticks: { color: '#8b949e' } } }, plugins: { legend: { labels: { color: '#e6edf3' } } } } } ); } } // ================================================================ // INICIALIZAÇÃO // ================================================================ window.onload = carregarGestaoEnergetica;
